-- Selezionare tutti gli studenti iscritti al Corso di Laurea in Economia

SELECT s.name,
       s.surname,
       s.fiscal_code,
       s.registration_number,
       s.enrolment_date,
       d.name AS corso_di_laurea
FROM students s
JOIN degrees d
  ON s.degree_id = d.id 
WHERE d.name = 'Corso di Laurea in Economia';

-- Selezionare tutti i Corsi di Laurea Magistrale del Dipartimento di 
Neuroscienze

SELECT deg.name AS degree_name,
       deg.level,
       deg.email,
       deg.website AS degree_website,
       deg.address,
       d.name AS department,
       d.website AS department_website
FROM degrees deg
JOIN departments d
  ON deg.department_id = d.id
WHERE d.name = 'Dipartimento di Neuroscienze'
  AND deg.level ='magistrale';


-- Selezionare tutti i corsi in cui insegna Fulvio Amato (id=44)

SELECT c.name AS name_course,
       t.name AS name_teacher,
       t.surname AS surname_teacher
FROM courses c
JOIN course_teacher ct
  ON c.id = ct.course_id
JOIN teachers t
  ON ct.teacher_id = t.id
WHERE t.id = 44;

-- Selezionare tutti gli studenti con i dati relativi al corso di laurea a cui 
sono iscritti e il relativo dipartimento, in ordine alfabetico per cognome e 
nome

SELECT s.surname,
       s.name,
       deg.name AS degree_name,
       deg.level,
       deg.email AS email_degree,
       deg.website AS website_degree,
       d.name AS department_name 
FROM students s  
JOIN degrees deg
  ON s.degree_id = deg.id
JOIN departments d
  ON deg.department_id = d.id
ORDER BY s.surname, s.name;    
     
-- Selezionare tutti i corsi di laurea con i relativi corsi e insegnanti

SELECT d.name AS degree_name,
	  c.name AS course_name,
	  t.name AS teacher_name,
	  t.surname AS teacher_surname
FROM degrees d
JOIN courses c
  ON d.id = c.degree_id
JOIN course_teacher ct
  ON c.id = ct.course_id
JOIN teachers t
  ON ct.teacher_id = t.id;

-- Selezionare tutti i docenti che insegnano nel Dipartimento di 
   Matematica (54)  

SELECT DISTINCT
       t.name,
       t.surname,
       dep.name AS department_name
FROM teachers t
JOIN course_teacher ct
  ON t.id = ct.teacher_id
JOIN courses c
  ON ct.course_id = c.id
JOIN degrees d
  ON c.degree_id = d.id
JOIN departments dep
  ON d.department_id = dep.id
WHERE dep.name = 'Dipartimento di Matematica';


-- BONUS: Selezionare per ogni studente il numero di tentativi sostenuti 
per ogni esame, stampando anche il voto massimo. Successivamente, 
filtrare i tentativi con voto minimo 18

SELECT
	  es.student_id,
    CONCAT(s.surname, ' ', s.name) AS studente,
    es.exam_id,
    COUNT(*) AS numero_tentativi,
    MAX(es.vote) AS voto_massimo
FROM exam_student es
JOIN students s
  ON s.id = es.student_id
GROUP BY
    s.id,
    s.name,
    s.surname,
	  es.exam_id
HAVING MAX(es.vote) >= 18
ORDER BY
    s.surname,
    s.name;
